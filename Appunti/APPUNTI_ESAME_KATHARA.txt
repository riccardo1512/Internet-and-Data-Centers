================================================================================
                                    RIP
================================================================================
# daemons: ripd=yes, zebra=yes

[ RIP BASE ]
router rip
    network 20.0.0.0/16             # Abilita RIP sulle interfacce nel prefisso
    redistribute bgp                # Inietta rotte BGP dentro RIP
    ! redistribute connected        # (Opzionale) inietta interfacce connesse
log file /var/log/frr/frr.log


[ RIP CON ANNUNCIO LOOPBACK (PER iBGP) ]
# Regola: se usi Loopback per iBGP, devi annunciarla in RIP altrimenti il vicino non sa raggiungerla!
# E' preferibile usare 'network <IP>/32' per evitare di spillare DMZ esterne dentro RIP:
router rip
    network 1.0.0.0/8
    network 1.1.1.1/32             # Annuncia SOLO la loopback (pulito, no route leaking)
    redistribute bgp
log file /var/log/frr/frr.log


[ RIP DETERMINISTICO (SOLO SE RICHIESTO: EVITARE RACE CONDITION / LOOP iBGP) ]
# Usato nei Transit AS per evitare che router interni rimbalzino traffico tra border.
# Inietta in RIP solo le rotte apprese da eBGP (next-hop esterno) e NON quelle iBGP:
ip prefix-list myNeighbors permit 12.0.0.0/30 le 32

route-map eBGPonly permit 10
    match ip next-hop prefix-list myNeighbors

router rip
    network 100.0.0.0/8
    redistribute connected
    redistribute bgp route-map eBGPonly
log file /var/log/frr/frr.log

router ospf

log file /var/log/frr/frr.log


[ OSPF CON ANNUNCIO LOOPBACK (PER iBGP) ]
# Se l'AS usa OSPF come IGP interno e si fa peering iBGP su Loopback:
# Basta dichiarare la propria loopback nell'area backbone di OSPF:
router ospf
    network 10.0.0.0/16 area 0.0.0.0       # Sottoreti fisiche interne AS
    network 1.1.1.1/32 area 0.0.0.0        # Annuncia la Loopback in OSPF (Type 1 LSA)
log file /var/log/frr/frr.log


[ OSPF ASBR (ROUTING STATICO ESTERNO) ]
# Definire la statica qui in frr.conf - NON con ip route add in .startup:
ip route 20.0.0.0/8 10.0.0.2

router ospf
    network 1.0.0.0/8 area 0.0.0.0
    redistribute static
log file /var/log/frr/frr.log


[ OSPF ABR (CONFINE TRA BACKBONE AREA 0 E AREA STUB) ]
router ospf
    network 10.0.0.0/16 area 0.0.0.0
    network 100.0.0.0/30 area 1.1.1.1
    area 1.1.1.1 stub               # Dichiara questa area come Stub
log file /var/log/frr/frr.log


[ OSPF ROUTER INTERNO AD AREA STUB ]
router ospf
    network 100.0.0.0/30 area 1.1.1.1
    network 200.0.0.0/24 area 1.1.1.1
    area 1.1.1.1 stub               # OBBLIGATORIO su TUTTI i router di questa area!
log file /var/log/frr/frr.log

                                    BGP

# daemons: bgpd=yes, zebra=yes

[ BGP STANDARD (PEERING SU LINK FISICO) ]
router bgp 100
    no bgp ebgp-requires-policy     # Decommentare se non ci sono filtri (FRR blocca eBGP senza filtri)
    no bgp network import-check     # Permette di annunciare prefissi senza match esatto in tabella IGP

    ! Vicino eBGP (AS differente):
    neighbor 11.0.0.2 remote-as 200
    neighbor 11.0.0.2 description Router eBGP

    ! Vicino iBGP (Stesso AS - link fisico):
    neighbor 200.1.0.2 remote-as 100
    neighbor 200.1.0.2 description Router iBGP interno
    neighbor 200.1.0.2 next-hop-self          # OBBLIGATORIO in iBGP!

    ! Inviare default route a un cliente/stub:
    neighbor 11.0.0.2 default-originate

    ! Prefisso proprio da annunciare:
    network 100.1.0.0/16
log file /var/log/frr/frr.log


[ iBGP SU LOOPBACK (update-source) ]
# Usato quando il compito dice: "AS uses loopbacks for its iBGP peerings".
# 1. In .startup: ip address add 1.1.1.1/32 dev lo
# 2. In RIP: network 1.1.1.1/32 (oppure in OSPF: network 1.1.1.1/32 area 0.0.0.0)
# 3. Sotto router bgp:
router bgp 100
    no bgp ebgp-requires-policy
    no bgp network import-check

    neighbor 2.2.2.2 remote-as 100
    neighbor 2.2.2.2 description Router iBGP su Loopback
    neighbor 2.2.2.2 update-source 1.1.1.1    # La propria loopback (oppure: update-source lo)
    neighbor 2.2.2.2 next-hop-self            # Riscrive il next-hop per il vicino interno

    network 100.1.0.0/16
log file /var/log/frr/frr.log

# NOTA D'ESAME: la Loopback si usa SOLO per iBGP (stesso AS).
# Tra AS differenti (eBGP) si usa SEMPRE l'IP fisico del cavo /30. Tra AS diversi non si usa mai RIP.


[ 1. BGP POLICIES: PREFIX-LIST (FILTRO PREFISSI IP) ]
# Sintassi definizioni:
ip prefix-list <NOME> permit 100.1.0.0/16
ip prefix-list <NOME> permit 200.1.0.0/16 le 24      # Da /16 fino a /24 comprese
ip prefix-list <NOME> permit 10.0.0.0/8 ge 16 le 24  # Maschere tra /16 e /24
ip prefix-list <NOME> deny 192.168.1.0/24            # Scarta prefisso
ip prefix-list <NOME> permit 0.0.0.0/0               # Solo la default route
ip prefix-list <NOME> permit any                     # Qualsiasi prefisso

# Applicazione sul vicino (in ingresso o in uscita):
router bgp 100
    neighbor 11.0.0.2 prefix-list <NOME> in
    neighbor 11.0.0.2 prefix-list <NOME> out


[ 2. BGP POLICIES: AS-PATH ACCESS-LIST (FILTRO PERCORSI AS) ]
# Sintassi definizioni:
bgp as-path access-list <NOME> permit <REGEXP>
bgp as-path access-list <NOME> deny <REGEXP>

# Applicazione sul vicino (filter-list):
router bgp 100
    neighbor 11.0.0.2 filter-list <NOME> in
    neighbor 11.0.0.2 filter-list <NOME> out
!
# --- TABELLA REGEX AS-PATH (MEMORIZZARE PER ESAME) ---
# Simboli:
#   ^  = inizio stringa AS-Path
#   $  = fine stringa AS-Path
#   _  = delimitatore (spazio, inizio, fine, virgola)
#   .  = qualsiasi singolo carattere
#   *  = 0 o più ripetizioni
#   +  = 1 o più ripetizioni
#   ?  = 0 o 1 ripetizione
#   [] = range di caratteri (es: [0-9])
#   |  = OR logico (es: 100|200)
#
# Pattern Tipici Esame:
#   ^$          Rotte originate LOCALMENTE (AS-Path vuoto)
#   _100_       Rotte che TRANSITANO per AS 100 (in qualsiasi posizione)
#   ^100_       Rotte RICEVUTE DIRETTAMENTE da AS 100 vicino
#   _100$       Rotte ORIGINATE da AS 100 (AS 100 all'estremita' destra)
#   ^100$       Rotte sia originate sia ricevute da AS 100 (lunghezza 1)
#   ^[0-9]+$    Rotte originate da QUALSIASI vicino diretto (lunghezza 1)
#   .*          QUALSIASI rotta (permette tutto)


[ 3. BGP POLICIES: ROUTE-MAP & ACCESS-LIST (MED, LOCAL-PREF, MANIPOLAZIONE) ]
# Access-List (usata per selezionare il prefisso da matchare nella route-map):
access-list myAggregate permit 100.1.0.0/16

# Route-Map Outbound: imposta MED (influenzare l'ingresso altrui, piu' BASSO vince):
route-map metricOut permit 10
    match ip address myAggregate
    set metric 10

# Route-Map Inbound: imposta Local-Pref (scegliere l'uscita locale, piu' ALTO vince, default 100):
route-map localPrefIn permit 10
    set local-preference 150

# Route-Map Deny: scarta cio' che matcha e fa passare il resto:
route-map filtroTotale deny 10
    match as-path <NOME_ACL>
route-map filtroTotale permit 20

# Applicazione sul vicino:
router bgp 100
    neighbor 11.0.0.2 route-map localPrefIn in
    neighbor 11.0.0.2 route-map metricOut out


[ ARCHITETTURA: BGP + RIP MULTI-HOMED (LOAD SHARING & REDISTRIBUZIONE) ]
# Border Router: daemons -> bgpd=yes, ripd=yes
router rip
    network 100.1.0.0/16
    redistribute bgp                # Inietta la default route BGP dentro RIP
!
router bgp 100
    no bgp network import-check
    neighbor 11.0.0.1 remote-as 10
    neighbor 11.0.0.1 prefix-list mineOut out
    neighbor 11.0.0.1 prefix-list defaultIn in

    # Load Sharing: Border 1 annuncia /17 bassa, Border 2 annuncia /17 alta, entrambi /16 backup
    network 100.1.0.0/16            # Backup globale
    network 100.1.0.0/17            # Primario prima meta - su Border 2 mettere: 100.1.128.0/17
!
ip prefix-list mineOut permit 100.1.0.0/16
ip prefix-list mineOut permit 100.1.0.0/17
ip prefix-list defaultIn permit 0.0.0.0/0


[ ARCHITETTURA: BGP + OSPF GATEWAY (DEFAULT ROUTE IN OSPF) ]
# Gateway Router: daemons -> bgpd=yes, ospfd=yes
interface eth1
    ospf cost 10
!
router ospf
    network 10.0.0.0/16 area 0.0.0.0
    default-information originate always   # Inietta la default route dentro OSPF
!
router bgp 100
    no bgp network import-check
    neighbor 11.0.0.1 remote-as 10
    neighbor 11.0.0.1 prefix-list defaultIn in
    network 10.0.0.0/16
!
ip prefix-list defaultIn permit 0.0.0.0/0

--- DOMANDE ESAME (answers.txt) ---
[ 1. Router ID di router X ]
vtysh -c "show ip ospf" | grep "Router ID"
# Regola: router-id manuale > max IP loopback > max IP eth fisiche attive

[ 2. Designated Router della LAN X ]
vtysh -c "show ip ospf interface eth0" | grep -A 2 "Designated Router"
# Regola: max priority (default 1) > max Router ID

[ 3. Quante 0/0 (default route) in Zebra su router X? ]
vtysh -c "show ip route" | grep -E "0\.0\.0\.0/0|default"
# Regola: router interno a Stub = 1 (o num ABR); router in Area 0 (Backbone) = 0

[ 4. BGP Decision Process (Come viene scelto il Best Path '>') ]
# 1. Weight (locale Cisco/FRR, piu' alto vince)
# 2. Local Preference (scambiata in iBGP, piu' ALTO vince, default 100)
# 3. Locally Originated (rotta da network/redistribute locale vince su appresa)
# 4. AS-Path Length (AS-Path piu' CORTO vince)
# 5. Origin Type (IGP [i] vince su EGP [e] che vince su Incomplete [?])
# 6. MED / Multi-Exit Discriminator (inviato tra AS adiacenti, piu' BASSO vince)
# 7. eBGP path vince su iBGP path
# 8. Minor costo IGP per raggiungere il BGP next-hop
# 9. Router-ID BGP piu' basso

[ 5. Stato sessione BGP con vicino X ]
vtysh -c "show ip bgp summary"
# Se sotto 'State/PfxRcd' c'e' un NUMERO (es: 0, 1, 5) -> sessione ESTABLISHED (UP)
# Se c'e' scritto 'Active' o 'Connect' -> TCP handshake fallito / rotta mancante
# Se c'e' scritto 'Idle' -> sessione disabilitata / amministrativamente giu'
